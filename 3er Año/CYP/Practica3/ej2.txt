
/* 2. Existen N personas que deben fotocopiar un documento cada una. Resolver cada ítem

por todas las personas, y las mismas la deben usar de a una persona a la vez, sin

Sólo se deben usar los procesos que representan a las Personas (y los monitores que sean
necesarios)*/

process persona[id: 0..N-1]{

}

/*b) Modifique la solución de (a) para el caso en que se deba respetar el orden de llegada.*/
process persona[id: 0..N-1]{
    fotocopiadora.solicitar(id);
    Fotocopiar();
    fotocopiadora.liberar();
}

Monitor fotocopiadora{
    cond espera[N];
    cola colaE;
    int cantE = 0;
    bool uso = false;
    Procedure solicitar(id: in int){
        if uso {
            cantE++;
            push(colaE,id);
            wait(espera[id]);
        }
        uso=true;
    }

    Procedure liberar(){
        int turno;
        if cantE>0 {
            pop(colaE,turno);
            cantE--;
            signal(espera[turno]);
        }
        else uso = false;
    }
}

/*e) Modifique la solución de (b) para el caso en que además haya un Empleado que le indica
a cada persona cuando debe usar la fotocopiadora.*/

process persona[id: 0..N-1]{
    fotocopiadora.solicitar(id);
    Fotocopiar();
    fotocopiadora.liberar();
}

process empleado{
    while true fotocopiadora.proximo();
}

Monitor fotocopiadora{
    cond espera[N], empleado;
    cola colaE;
    int cantE = 0;
    bool uso = false;
    Procedure solicitar(id: in int){
        cantE++;
        push(colaE,id);
        wait(espera[id]);
    }

    Procedure liberar(){
        uso = false;
        signal(empleado);
    }

    Procedure proximo(){
        if cantE>0 and not uso {
            int turno;
            pop(colaE,turno);
            cantE--;
            signal(espera[turno]);
            uso = true;
        } else wait(empleado);
    }
}

/*f) Modificar la solución (e) para el caso en que sean 10 fotocopiadoras. El empleado le
indica a la persona cuando puede usar una fotocopiadora, y cual debe usar */

process persona[id: 0..N-1]{
    fotocopiadora.solicitar(id);
    Fotocopiar();
    fotocopiadora.liberar(id);
}

process empleado{
    while true fotocopiadora.proximo();
}

Monitor fotocopiadora{
    cond espera[N], empleado;
    cola colaE;
    int cantE = 0, disp = 10;
    bool fot[10] = ([10] false);
    int impAsignada[N];

    Procedure solicitar(id: in int){
        cantE++;
        push(colaE,id);
        signal(empleado);
        wait(espera[id]);
    }

    Procedure liberar(id: in int){
        fot[impAsignada[id]] = false;
        disp++;
        signal(empleado);
    }

    Procedure proximo(){
        if cantE>0 and disp>0  {
            int turno, idD = 0;
            pop(colaE,turno);
            cantE--;
            while fot[idD] idD++;
            impAsignada[turno] = idD;
            disp--;
            fot[idD] = true;
            signal(espera[turno]);
        } else wait(empleado);
    }
}
