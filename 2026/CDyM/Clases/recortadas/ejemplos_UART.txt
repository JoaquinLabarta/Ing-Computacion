
 //1-transmitir 'G'

#include <avr/io.h>
#define F_CPU 16000000UL		// 16 MHz
#include <util/delay.h>

 int main(void)
 {
	 //initialize the USART
	 UCSR0B = (1<<TXEN0);				// Activa el Transmisor del Puerto Serie
	 UCSR0C = (1<<UCSZ01)|(1<<UCSZ00);	 //configura formato 8N1
	 UBRR0L = 103;						 //configura BAUDRATE = 9600bps@16MHz
	 while(1) {
		 while (! (UCSR0A & (1<<UDRE0)));  //esperar a que UDR0 este vacio
		 UDR0 = 'G';						//transmitir �G� 
		 _delay_ms(100);
	 }
	 return 0;
 }
